#!/bin/bash
export DEBIAN_FRONTEND=noninteractive
set -euo pipefail

echo "[PROVISION] apt update"
apt-get update -y

echo "[PROVISION] install core packages + Java"
apt-get install -y --no-install-recommends \
  python3 python3-pip python3-venv \
  curl git build-essential unzip sudo \
  bash vim docker.io awscli \
  ca-certificates \
  openjdk-17-jre-headless

mkdir -p /usr/local/bin

# --- HARD GUARANTEE: java exists on PATH ---
echo "[PROVISION] verify java"
if ! command -v java >/dev/null 2>&1; then
  echo "[PROVISION] java not in PATH, attempting to link it..."
  # common Debian/Ubuntu location:
  if [ -x /usr/lib/jvm/java-17-openjdk-amd64/bin/java ]; then
    ln -sf /usr/lib/jvm/java-17-openjdk-amd64/bin/java /usr/bin/java
    ln -sf /usr/lib/jvm/java-17-openjdk-amd64/bin/java /usr/local/bin/java
  fi
fi

if ! command -v java >/dev/null 2>&1; then
  echo "[PROVISION][ERROR] Java still missing after install/link."
  echo "Debug info:"
  ls -la /usr/bin/java || true
  find /usr/lib/jvm -type f -name java 2>/dev/null | head -n 20 || true
  exit 1
fi

java -version

# -------------------------------------------------
# NEXTFLOW INSTALL
# -------------------------------------------------
echo "[PROVISION] install Nextflow"
curl -s https://get.nextflow.io | bash
mv nextflow /usr/local/bin/nextflow
chmod +x /usr/local/bin/nextflow

# sanity check
nextflow -version || true

# Pre-create cache dirs to avoid permission issues
mkdir -p /root/.nextflow /root/.nextflow/assets /root/.nextflow/plugins
# -------------------------------------------------

echo "[PROVISION] install fetch helper"
cat >/tmp/fetch <<'EOF'
#!/bin/bash
set -euo pipefail

if [ $# -lt 1 ]; then
  echo "Usage: fetch <object-key> [dest_dir]"
  exit 1
fi

OBJ="$1"
DEST="${2:-/data}"

[[ "$OBJ" != */* ]] && OBJ="uploads/$OBJ"

mkdir -p "$DEST"

export AWS_EC2_METADATA_DISABLED=true
export AWS_ACCESS_KEY_ID=minioadmin
export AWS_SECRET_ACCESS_KEY=minioadmin
export AWS_DEFAULT_REGION=us-east-1

aws --endpoint-url http://minio:9000 \
  s3 cp "s3://cassie-bucket/$OBJ" "$DEST/"
EOF

chmod +x /tmp/fetch
install -m 755 /tmp/fetch /usr/local/bin/fetch
rm -f /tmp/fetch

echo "[PROVISION] install tool wrappers"
mkdir -p /tmp/dockerized_tools
cp -r /opt/dockerized_tools/* /tmp/dockerized_tools/
find /tmp/dockerized_tools -name "*.sh" -exec sed -i 's/\r$//' {} \;

for f in /tmp/dockerized_tools/run*.sh; do
  [ -f "$f" ] || continue
  install -m 755 "$f" "/usr/local/bin/$(basename "$f" .sh)"
done

echo "[PROVISION] docker permissions"
groupadd -f docker || true
chgrp docker /var/run/docker.sock 2>/dev/null || true
chmod g+rw /var/run/docker.sock 2>/dev/null || true

echo "[PROVISION] docker info"
docker info

echo "[PROVISION] build tool images"
cd /tmp/dockerized_tools
bash ./buildtools.sh

echo "[PROVISION] DONE"
