#!/bin/bash
export DEBIAN_FRONTEND=noninteractive
set -euxo pipefail

# Function to write progress to both stdout (for exec_run) and container logs (for Docker Desktop)
log_to_container() {
    echo "$@"  # Normal stdout (captured by exec_run)
    # Also write to container's main stdout so Docker Desktop can see it
    echo "$@" > /proc/1/fd/1 2>/dev/null || true
}

log_to_container "=========================================="
log_to_container "[PROVISION] Starting VM provisioning..."
log_to_container "=========================================="
log_to_container "[PROVISION] Step 1/6: Updating package lists..."
apt-get update -y

log_to_container "[PROVISION] Step 2/6: Installing core packages and Java..."
apt-get install -y --no-install-recommends \
  python3 python3-pip python3-venv \
  curl git build-essential unzip sudo \
  bash vim docker.io awscli \
  ca-certificates \
  openjdk-17-jre-headless

mkdir -p /usr/local/bin

# --- HARD GUARANTEE: java exists on PATH ---
log_to_container "[PROVISION] Step 2.1/6: Verifying Java installation..."
if ! command -v java >/dev/null 2>&1; then
  log_to_container "[PROVISION] java not in PATH, attempting to link it..."
  # common Debian/Ubuntu location:
  if [ -x /usr/lib/jvm/java-17-openjdk-amd64/bin/java ]; then
    ln -sf /usr/lib/jvm/java-17-openjdk-amd64/bin/java /usr/bin/java
    ln -sf /usr/lib/jvm/java-17-openjdk-amd64/bin/java /usr/local/bin/java
  fi
fi

if ! command -v java >/dev/null 2>&1; then
  log_to_container "[PROVISION][ERROR] Java still missing after install/link."
  echo "Debug info:"
  ls -la /usr/bin/java || true
  find /usr/lib/jvm -type f -name java 2>/dev/null | head -n 20 || true
  exit 1
fi

java -version

# -------------------------------------------------
# NEXTFLOW INSTALL
# -------------------------------------------------
log_to_container "[PROVISION] Step 3/6: Installing Nextflow..."
curl -s https://get.nextflow.io | bash
mv nextflow /usr/local/bin/nextflow
chmod +x /usr/local/bin/nextflow

# sanity check
nextflow -version || true

# Pre-create cache dirs to avoid permission issues
mkdir -p /root/.nextflow /root/.nextflow/assets /root/.nextflow/plugins
# -------------------------------------------------

log_to_container "[PROVISION] Step 4/6: Installing fetch helper..."
cat >/tmp/fetch <<'EOF'
#!/bin/bash
set -euo pipefail

if [ $# -lt 1 ]; then
  echo "Usage: fetch <object-key> [dest_dir]"
  exit 1
fi

OBJ="$1"
DEST="${2:-/data}"

[[ "$OBJ" != */* ]] && OBJ="uploads/$OBJ"

mkdir -p "$DEST"

export AWS_EC2_METADATA_DISABLED=true
export AWS_ACCESS_KEY_ID=minioadmin
export AWS_SECRET_ACCESS_KEY=minioadmin
export AWS_DEFAULT_REGION=us-east-1

aws --endpoint-url http://minio:9000 \
  s3 cp "s3://cassie-bucket/$OBJ" "$DEST/"
EOF

chmod +x /tmp/fetch
install -m 755 /tmp/fetch /usr/local/bin/fetch
rm -f /tmp/fetch

log_to_container "[PROVISION] Step 5/6: Installing tool wrappers..."
# Create wrapper scripts that call scripts directly from /opt/dockerized_tools
# This allows updating scripts on host without reprovisioning
for script in /opt/dockerized_tools/run*.sh; do
  [ -f "$script" ] || continue
  script_name=$(basename "$script" .sh)
  wrapper="/usr/local/bin/$script_name"
  cat > "$wrapper" <<EOF
#!/bin/bash
# Wrapper that calls script from mounted dockerized_tools directory
# This allows script updates without reprovisioning
exec bash "/opt/dockerized_tools/${script_name}.sh" "\$@"
EOF
  chmod +x "$wrapper"
  log_to_container "[PROVISION]   Created wrapper: $wrapper -> $script"
done

log_to_container "[PROVISION] Step 5.1/6: Setting up Docker permissions..."
groupadd -f docker || true
chgrp docker /var/run/docker.sock 2>/dev/null || true
chmod g+rw /var/run/docker.sock 2>/dev/null || true

log_to_container "[PROVISION] Step 5.2/6: Verifying Docker access..."
docker info

log_to_container "[PROVISION] Step 6/6: Verifying Docker tool images are available..."
log_to_container "[PROVISION] Note: Tool images should be pre-built on the host before provisioning"
log_to_container "[PROVISION] Checking for required tool images..."

# Check if tool images exist (they should be built on host and available via shared Docker socket)
MISSING_IMAGES=0

check_image() {
    local image_name="$1"
    # Check if image exists (with or without tag)
    # docker images output format: REPOSITORY   TAG
    # Images without explicit tag show as "REPO   <none>" or "REPO   latest"
    if docker images --format "{{.Repository}}:{{.Tag}}" | grep -qE "^${image_name}$|^${image_name}:latest$" || \
       docker images --format "{{.Repository}}" | grep -qE "^${image_name}$"; then
        log_to_container "[PROVISION] ✓ Found image: $image_name"
        return 0
    else
        log_to_container "[PROVISION] ✗ Missing image: $image_name"
        MISSING_IMAGES=$((MISSING_IMAGES + 1))
        return 1
    fi
}

check_image "fastqc:0.12.1"
check_image "genomescope2"
check_image "spades"
check_image "quast"

if [ $MISSING_IMAGES -gt 0 ]; then
    log_to_container "[PROVISION] ERROR: $MISSING_IMAGES required tool image(s) are missing!"
    log_to_container "[PROVISION] Please build tool images on the host first:"
    log_to_container "[PROVISION]   cd dockerized_tools && bash buildtools.sh"
    log_to_container "[PROVISION] Then restart provisioning."
    exit 1
fi

log_to_container "[PROVISION] All required tool images are available."

log_to_container "=========================================="
log_to_container "[PROVISION] ✓ Provisioning completed successfully!"
log_to_container "[PROVISION] VM is ready for tenant containers"
log_to_container "=========================================="
